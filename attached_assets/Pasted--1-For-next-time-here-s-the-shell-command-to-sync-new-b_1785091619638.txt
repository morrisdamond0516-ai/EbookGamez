2.....(Here's a single copy-paste command you can run anytime in the Replit shell to sync all missing books from the live site)(What it does step by step: a.Fetches all book IDs from the live site   b. Gets all book IDs from your local database   c. Finds which ones are missing locally    d.  Fetches each missing book from the live API     e. Inserts them all into your local database   f.  Prints the final count so you can confirm it worked).

curl -s "https://EbookGamez.replit.app/api/books?limit=1000" | grep -o '"id":[0-9]*' | sed 's/"id"://' | sort -n > /tmp/live_ids.txt && \
psql "$DATABASE_URL" -c "SELECT id FROM books ORDER BY id;" -t | tr -d ' ' | grep -v '^$' | sort -n > /tmp/local_ids.txt && \
MISSING=$(comm -23 /tmp/live_ids.txt /tmp/local_ids.txt | tr '\n' ' ') && \
echo "Missing IDs: $MISSING" && \
echo "BEGIN;" > /tmp/sync.sql && \
for id in $MISSING; do
  curl -s "https://EbookGamez.replit.app/api/books/$id" > /tmp/bk.json
  TITLE=$(cat /tmp/bk.json | grep -o '"title":"[^"]*"' | head -1 | sed 's/"title":"//;s/"$//' | sed "s/'/''/g")
  GENRE=$(cat /tmp/bk.json | grep -o '"genre":"[^"]*"' | head -1 | sed 's/"genre":"//;s/"$//' | sed "s/'/''/g")
  CAT=$(cat /tmp/bk.json | grep -o '"category":"[^"]*"' | head -1 | sed 's/"category":"//;s/"$//' | sed "s/'/''/g")
  PRICE=$(cat /tmp/bk.json | grep -o '"price":"[^"]*"' | head -1 | sed 's/"price":"//;s/"$//')
  RATING=$(cat /tmp/bk.json | grep -o '"rating":"[^"]*"' | head -1 | sed 's/"rating":"//;s/"$//')
  COVER=$(cat /tmp/bk.json | grep -o '"coverUrl":"[^"]*"' | head -1 | sed 's/"coverUrl":"//;s/"$//')
  VISIBLE=$(cat /tmp/bk.json | grep -o '"visible":[a-z]*' | head -1 | sed 's/"visible"://')
  SRC=$(cat /tmp/bk.json | grep -o '"sourceDraftId":[0-9]*' | head -1 | sed 's/"sourceDraftId"://')
  CREATED=$(cat /tmp/bk.json | grep -o '"createdAt":"[^"]*"' | head -1 | sed 's/"createdAt":"//;s/"$//')
  [ -z "$SRC" ] && SRC="NULL"
  echo "INSERT INTO books (id,title,author,genre,category,price,rating,cover_url,visible,cover_fit,source_draft_id,created_at) VALUES ($id,'$TITLE','EbookGamez','$GENRE','$CAT',$PRICE,$RATING,'$COVER',$VISIBLE,'cover',$SRC,'$CREATED') ON CONFLICT (id) DO NOTHING;" >> /tmp/sync.sql
done && \
echo "COMMIT;" >> /tmp/sync.sql && \
psql "$DATABASE_URL" -f /tmp/sync.sql && \
psql "$DATABASE_URL" -c "SELECT COUNT(*) AS total_books, MAX(id) AS latest_id FROM books;"
